cmake_minimum_required(VERSION 3.28)
project(ethercat_transport_checks C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(ec_transport STATIC
  "${REPO}/src/transport/ec_transport.c"
  "${REPO}/src/transport/ec_linux_transport.c")
target_include_directories(ec_transport PUBLIC "${REPO}/src/transport")
target_compile_options(ec_transport PRIVATE -Wall -Wextra -Werror)
add_executable(test_transport test_transport.c)
target_link_libraries(test_transport PRIVATE ec_transport)
target_compile_options(test_transport PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME linux_transport COMMAND test_transport)
set_tests_properties(linux_transport PROPERTIES TIMEOUT 15)
if(SOEM_SOURCE)
  set(SOEM_BUILD_SAMPLES ON CACHE BOOL "" FORCE)
  add_subdirectory("${SOEM_SOURCE}" soem)
  target_sources(soem PRIVATE
    "${REPO}/src/transport/ec_transport.c"
    "${REPO}/src/transport/ec_linux_transport.c")
  target_include_directories(soem PUBLIC "$<BUILD_INTERFACE:${REPO}/src/transport>")
  add_executable(test_soem_transport test_soem_transport.c)
  target_link_libraries(test_soem_transport PRIVATE soem)
  target_compile_options(test_soem_transport PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME soem_transport COMMAND test_soem_transport)
  set_tests_properties(soem_transport PROPERTIES TIMEOUT 15)
endif()
